Status Groups & Description
The statuses are grouped into 3 pipeline stages:
GroupLabelMeaningÀ faireProspectLead identified, no action taken yetEn coursContactéFirst contact has been madeEn coursVisite à planifierVisit/meeting needs to be scheduledEn coursVisite planifiéeVisit/meeting is confirmed and scheduledEn coursOffre effectuéeA proposal/offer has been sentEn coursOffre rejetéeThe proposal was declinedTerminéOffre acceptéeThe proposal was accepted — deal won ✅

Enum
tsexport enum ProspectStatus {
  PROSPECT          = 'PROSPECT',
  CONTACTE          = 'CONTACTE',
  VISITE_A_PLANIFIER = 'VISITE_A_PLANIFIER',
  VISITE_PLANIFIEE  = 'VISITE_PLANIFIEE',
  OFFRE_EFFECTUEE   = 'OFFRE_EFFECTUEE',
  OFFRE_REJETEE     = 'OFFRE_REJETEE',
  OFFRE_ACCEPTEE    = 'OFFRE_ACCEPTEE',
}

export enum StatusGroup {
  A_FAIRE  = 'À faire',
  EN_COURS = 'En cours',
  TERMINE  = 'Terminé',
}

CSS / Tailwind Badge Component
tsx// StatusBadge.tsx
const StatusBadge = ({ status }: { status: ProspectStatus }) => {
  const config = STATUS_CONFIG[status];
  return (
    <span
      className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-sm font-medium"
      style={{ backgroundColor: config.bg, color: config.text }}
    >
      <span
        className="w-2 h-2 rounded-full flex-shrink-0"
        style={{ backgroundColor: config.dot }}
      />
      {config.label}
    </span>
  );
};
